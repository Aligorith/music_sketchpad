Yes. The solution is to maintain a separate accidental-memory state in JavaScript, independent of the LilyPond notation.

The LilyPond text should contain the actual pitch names, while JavaScript remembers whether a pitch has been explicitly naturalised during the current bar.

For your example, after `cis''4` is lowered to `c''4`, the next C should remain natural.

## 1. Add an explicit accidental override state

Add this alongside your application state:

javascript

```javascript

// Key format: letter:octave

```

## 2. Add a helper to reconstruct accidental state

This reads the notation and incorporates the explicit overrides.

javascript

```javascript
function getAccidentalState() {

  const state = {};

  const tokens = tokenize(
    parts[selectedPart].notation
  );

  for(const raw of tokens) {

    const token = parseToken(raw);

    if(token.bar) {
      Object.keys(state).forEach(
        key => delete state[key]
      );
      continue;
    }

    if(token.rest) continue;

    pitchInfo(token,state);
  }

  // Apply explicit naturalisations and other overrides.
  Object.entries(accidentalOverrides).forEach(
    ([key,value]) => {
      state[key] = value;
    }
  );

  return state;
}
```

## 3. Replace `appendPitch()`

javascript

```javascript
function appendPitch(letter) {

  const tokens = tokenize(
    parts[selectedPart].notation
  );

  let lastPitch = null;

  for(let i=tokens.length-1;i>=0;i--) {

    if(tokens[i] === "|" || /^r/.test(tokens[i])) {
      continue;
    }

    try {
      const token = parseToken(tokens[i]);
      lastPitch = pitchInfo(token,{});
      if(lastPitch) break;
    } catch(e) {}
  }

  const semitones = {
    c:0,d:2,e:4,f:5,g:7,a:9,b:11
  };

  let octave = 3;

  if(lastPitch) {

    const candidates = [];

    for(let o=lastPitch.octave-1;
        o<=lastPitch.octave+1;o++) {

      candidates.push({
        octave:o,
        pitch:12*(o+1)+semitones[letter]
      });
    }

    candidates.sort((a,b)=>
      Math.abs(a.pitch-lastPitch.pitch)-
      Math.abs(b.pitch-lastPitch.pitch)
    );

    octave=candidates[0].octave;
  }

  const key=letter+":"+octave;

  const state=getAccidentalState();

  const accidental=state[key]||"";

  const marks=octave>3
    ? "'".repeat(octave-3)
    : ",".repeat(3-octave);

  const pitch=letter+accidental+marks;

  previewPitch(pitch);

  appendToken(
    pitch+durationText(currentDuration)
  );
}
```

## 4. Replace `updateLastNote()`

This is the important change. When a semitone adjustment produces a natural note, it records that naturalisation separately.

javascript

```javascript
function updateLastNote(transform) {

  const p=parts[selectedPart];
  const tokens=tokenize(p.notation);

  const index=lastNoteIndex();

  if(index<0) return;

  const token=parseToken(tokens[index]);

  const state=getAccidentalState();
  const info=pitchInfo(token,state);

  if(!info) return;

  const newMidi=transform(info.pitch);

  const converted=midiToLilyWithAccidental(newMidi);

  const newPitch=
    converted.name+
    converted.marks;

  // Remember the accidental independently of LilyPond.
  const key=converted.letter+":"+converted.octave;

  const standardAccidental=converted.accidental;

  accidentalOverrides[key]=standardAccidental;

  // Update notation without adding natural markers.
  tokens[index]=
    newPitch+
    token.duration+
    (token.dotted?".":"");

  p.notation=tokens.join(" ");

  $("notation").value=p.notation;

  renderScore();

  previewPitch(newPitch);
}
```

## 5. Reset overrides at barlines

Update `appendBar()`:
